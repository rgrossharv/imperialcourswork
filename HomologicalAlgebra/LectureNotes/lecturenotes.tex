\subsection*{Basics of Category Theory}

\begin{definition}
    A \emph{category} $\mathcal{C}$ consists of 
    \begin{enumerate}
        \item a collection of `objects' called $\text{ob}(\mathcal{C})$
        \item $\forall A, B \in \text{ob}(\mathcal{C})$ we have a set of ``morphisms'' $\text{Hom}_{\mathcal{C}}(A,B)$
        \item An identity (or a unit) for each object $\text{id}_A \in \text{Hom}_{\mathcal{C}}(A,A)$
        \item A composition rule for $A,B,C$ which is: $\text{Hom}_{\mathcal{C}}(B,C) \times \text{Hom}_{\mathcal{C}}(A,B) \to \text{Hom}_{\mathcal{C}}(A,C)$ It is simple to think of $g \circ f \in \text{Hom}_{\mathcal{C}}(A,C)$ where $g \in \text{Hom}_{\mathcal{C}}(B,C)$ and $f \in \text{Hom}_{\mathcal{C}}(A,B)$ so that we really have composition of functions here. 
        \item These functions are also associative such that for $f \in \text{Hom}_{\mathcal{C}}(A,B), g \in \text{Hom}_{\mathcal{C}}(B,C), h \in \text{Hom}_{\mathcal{C}}(C,D)$ we have $h \circ (g \circ f) = (h \circ g) \circ f$ and the same for units $\text{id}_B \circ f = f \circ \text{id}_A = f$.
    \end{enumerate}
\end{definition}

Now, we move onto subcategories which is exactly what you think it is. 

\begin{definition}
    A \emph{subcategory} of a category, $\CatD \subseteq \Cat$ is a subcollection $\ob(\CatD) \subseteq \ob(\Cat)$ and we have for all $X, Y \in \ob(\CatD)$ we have $\Hom_\CatD(X,Y) \subseteq \Hom_\Cat(X,Y)$ such that \begin{enumerate}
        \item $\text{id}_X \in \Hom_\CatD(X,X)$
        \item It is closed under composition (Need more substance here).
    \end{enumerate} 
\end{definition}
\noindent
We call $\CatD$ a full subcategory of $\Cat$ if $\forall X,Y \in \ob(\CatD)$ we have $\Hom_\CatD(X,Y) = \Hom_\Cat(X,Y)$. 

\begin{example}
    The category of abelian groups. We will call this $\CatD = \text{Ab}$ We have $\text{Ab} \hookrightarrow \text{Groups}$ is a full subcategory since the fact that two groups are abelian doesn't change the homomorphisms between them. Hence, $\Hom_\text{Ab}(A,B) = \Hom_\text{Groups}(A,B)$ for any choice $A,B \in \ob(\text{Ab})$.
\end{example}

\begin{example}
    For a field $k$, we consider the category of vector spaces of $k$, $\Vect_k = \Cat$, then we can have the sub-category of finite dimensional vector spaces of $k$, $\text{fd}(\Vect_k) = \CatD, \CatD \subseteq \Cat$ is a full subcategory.
\end{example}

\noindent
Now, moving ahead with a view towards functors, we need to introduce some machinery to understand some properties we will learn about later like contravariance. 

\begin{definition}
    if $\Cat$ is a category then $\Cat^\text{op}$ is the opposite category defined such that $\ob(\Cat) = \ob(\Cat^\text{op})$ where $\Hom_{\Cat^\text{op}}(X,Y) = \Hom_\Cat(Y,X)$.
\end{definition}

\begin{definition}
    A (covariant) \emph{functor} on two categories $\Cat$ and $\CatD$, $F: \Cat \to \CatD$ is an assignment between $\ob(\Cat) \to \ob(\CatD)$ that takes $X \in \ob(\Cat)$ to  $F(X) \in \ob(\CatD)$ so we have $F$ taking $X \to F(X)$ such that for any $X,Y \in \ob(\Cat)$ we know there exists $F_{X,Y}: \Hom_\Cat(X,Y) \to \Hom_\CatD(F(X),F(Y))$ where $F(f):F(X) \to F(Y)$ which also must have the properties that $F(\text{id}_X) = \text{id}_{F(X)}$ and that $F(f \circ g) = F(f) \circ F(g)$.
\end{definition}

\begin{definition}
    A \emph{contravariant functor} is a covariant functor $F: \Cat^{\text{op}} \to \CatD$ This means it flips the direction when we apply it to spaces and functions between their homomorphisms $\Hom_\Cat(X,Y) \to \Hom_\CatD(F(Y),F(X))$
\end{definition}

\begin{example}
    Here we will let $\Cat = \text{Top}$ and $\CatD = \text{Set}$ and we take the natural function from a topological space to the set underneath it. $F: \text{Top} \to \text{Set}$ which forgets the topology and just views the former topological space as a set. The precise way to think of this is that \begin{enumerate}
        \item $F(X)$ is merely the set $X$ where earlier it had a topology.
        \item $f \in \Hom_\text{Top}(X,Y)$ is continuous since all the morphisms are continuous but $F(f)$ is merely the function, it forgets that it is continuous.
    \end{enumerate}
\end{example}

\begin{definition}
    A functor $F$, $F:\Cat \to \CatD$ is called fully faithful if $f: Hom_\Cat(X,Y) \to \Hom_\CatD(F(X),F(Y))$ is bijective for $X,Y$ in $\Cat$
\end{definition}

\noindent
From now on $\ob(\Cat)$ will be replaced by just $\Cat$.

\begin{example}
    For any category $\Cat$ and any object $A \in \Cat$ there is a contravariant functor $\Cat \to \text{Set}$ where $F = (\Hom_\Cat(-,A))$ where the dash is where we will take an input for $F$. Hence, for $B \in \Cat$ we have $F(B) = \Hom_\Cat(B,A)$ and for $f \in \Hom_\Cat(B,B')$ we get $F(f) \in \Hom_\Cat(F(B'),F(B))$. This is a funciton $\Hom_\Cat(B', A) \to Hom_\Cat(B,A)$. Functors like this example are called representable and are represented by $A$.
\end{example}
